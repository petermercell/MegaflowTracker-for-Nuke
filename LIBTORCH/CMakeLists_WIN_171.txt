# ============================================================================
# CMakeLists.txt  —  MegaFlowSolve   *** WINDOWS / NUKE 17.1 build ***
#                    (Nuke 17.1v1 / libtorch 2.7.1 / CUDA 12.x / OpenCV, MSVC)
#
# Windows port of the Linux CMakeLists.txt. Same structure — a static
# megaflow_core, the MegaFlowSolve Nuke plugin, and the optional mf_solve CLI —
# with the parts that cannot cross over:
#
#  1. LIBTORCH.  Linux links Nuke's own libtorch*.so for ABI parity. Foundry
#     ships the torch DLLs on Windows but no import libraries, so this links a
#     standalone libtorch 2.7.1 (Release, cu128) via -DTORCH_DIR. At runtime the
#     loader resolves torch.dll / c10.dll etc. BY NAME and Nuke's own directory
#     is on the search path, so inside Nuke the plugin binds to Nuke's already
#     loaded copy. That only holds if TORCH_DIR is EXACTLY Nuke's version, 2.7.1.
#     The five import libs are linked by hand rather than through libtorch's
#     TorchConfig.cmake, which pulls in Caffe2's config: it wants
#     CUDA::nvToolsExt (gone in CUDA >= 12.8) and re-resolves the toolkit via
#     legacy FindCUDA, often landing on a different one.
#
#  2. /MD IS MANDATORY HERE. DDImage and libtorch are both /MD; a static CRT
#     would give this plugin its own heap and crash on the first allocation that
#     crosses the boundary. (The TensorRT BiRefNet plugins can use /MT because
#     they link no /MD C++ library — this one cannot.)
#
#  3. NO RPATH, no dl/rt/pthread, and DDImage.lib instead of libDDImage.so.
#
# OpenCV is still needed: MegaFlowSolve.h includes opencv2/opencv.hpp and the
# node uses cv::resize to match the CLI's resampling exactly. core + imgproc are
# enough for the plugin; the CLI additionally needs imgcodecs for EXR. Point
# CMake at your build with -DOpenCV_DIR=... (the folder holding
# OpenCVConfig.cmake, e.g. C:/opencv/build or a vcpkg share/opencv4 dir).
#
# The 5 .pt engines are NOT embedded — the node takes an "Engines Dir" knob and
# falls back to the MEGAFLOW_ENGINES environment variable, so nothing about the
# engines is baked into this build.
#
# Configure / build:
#   set PATH=C:\Program Files\NVIDIA GPU Computing Toolkit\CUDA\v12.9\bin;%PATH%
#   cd C:\Users\WORKSTATION\Downloads\MegaflowTracker-for-Nuke-main\MegaflowTracker-for-Nuke-main\LIBTORCH
#   copy /y CMakeLists_WIN_171.txt CMakeLists.txt
#   if exist build rd /s /q build 2>nul
#   if not exist build md build
#   cd /d build
#   cmake .. -G Ninja -DCMAKE_BUILD_TYPE=Release ^
#         -DNUKE_VERSION=17.1v1 ^
#         -DTORCH_DIR=C:/libtorch-2.7.1 ^
#         -DOpenCV_DIR=C:/opencv/build
#   cmake --build .
#   copy /y MegaFlowSolve.dll "%USERPROFILE%\.nuke\CPP171\"
#
# Also copy MegaFlowTracker.tcl and megaflow_cache.py from the repo root next to
# the DLL, and put the engines folder somewhere the knob (or MEGAFLOW_ENGINES)
# can point at.
# ============================================================================
cmake_minimum_required(VERSION 3.18)

# Must be set BEFORE project() for the MSVC runtime selection to take effect.
cmake_policy(SET CMP0091 NEW)

project(megaflow_solve VERSION 0.1.0 LANGUAGES CXX)

    message(FATAL_ERROR "This is the Windows CMakeLists — use the stock CMakeLists.txt on Linux.")

    message(FATAL_ERROR "This build requires MSVC (Visual Studio 2022 or later).")
endif()

    message(FATAL_ERROR "Visual Studio 2022 or later required (MSVC 19.30+), got ${MSVC_VERSION}")
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)

# /MD — not negotiable, see note 2 above.

#-------------------------------------------------------------------------------
# Common settings (mirrors mf_common from the Linux file)
#-------------------------------------------------------------------------------
set(COMMON_INCLUDES
    ${SRC}
    ${CMAKE_SOURCE_DIR}/include
    ${TORCH_INCLUDE_DIRS}
    ${CUDAToolkit_INCLUDE_DIRS}
    ${OpenCV_INCLUDE_DIRS})

function(mf_common tgt)
    target_include_directories(${tgt} PRIVATE ${COMMON_INCLUDES})

    target_compile_definitions(${tgt} PRIVATE
        NOMINMAX                 # windows.h would otherwise eat std::min/std::max
        WIN32_LEAN_AND_MEAN
        _USE_MATH_DEFINES
        _CRT_SECURE_NO_WARNINGS  # std::getenv & co.
        _SCL_SECURE_NO_WARNINGS
        _SILENCE_ALL_CXX17_DEPRECATION_WARNINGS

        # VS 17.10+ made std::mutex's ctor constexpr. Built that way, a /MD
        # binary access-violates inside an msvcp140.dll older than 14.40 — and
        # Nuke's own directory wins the DLL search (17.0v1 ships 14.36, 17.1v1
        # ships 14.44). libtorch's headers use std::mutex all over, and /MD is
        # mandatory here, so this one matters. Kept in both builds so a single
        # binary works in either Nuke.
        _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)

    target_compile_options(${tgt} PRIVATE
        /W3 /O2 /MP /bigobj /EHsc /permissive- /Zc:__cplusplus /nologo
        /wd4251 /wd4267 /wd4244 /wd4275 /wd4305 /wd4996 /wd4018)
endfunction()

#   cuda_utils.cpp compiles in here only — the node has the VRAM buttons.
#-------------------------------------------------------------------------------

    message(FATAL_ERROR "src/MegaFlowSolve.cpp not found — nothing to build")

    ${SRC}/MegaFlowSolve.cpp
    ${SRC}/cuda_utils.cpp)

target_include_directories(MegaFlowSolve PRIVATE ${NUKE_INCLUDE})
target_link_libraries(MegaFlowSolve PRIVATE
    megaflow_core
    ${DDIMAGE_LIBRARY}
    ${OpenCV_LIBS})
mf_common(MegaFlowSolve)

set_target_properties(MegaFlowSolve PROPERTIES
    PREFIX ""
    SUFFIX ".dll"
    OUTPUT_NAME "MegaFlowSolve")

    mf_common(mf_solve)
    message(STATUS "CLI: mf_solve ON — needs the torch DLLs and OpenCV DLLs on PATH,")
    message(STATUS "     and OPENCV_IO_ENABLE_OPENEXR=1 in the environment for EXR input.")
